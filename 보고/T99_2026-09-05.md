### 춘화는 달력에게 물어본다 — **수가 한 개도 없다**

```js
function _nextSeasonDay(day) {            // 그 계절의 다음 날 — calendarOf 가 답한다
  const cal = _events().calendarOf(_day(day));
  return _day(day) + Math.max(1, (cal.seasonDays | 0) - (cal.dayOfSeason | 0) + 1);
}
function _vernalCompute(id, p) {          // 심은 뒤 처음 만나는 겨울 → 그 겨울이 끝난 다음 날
  let d = p;
  for (let hop = 0, cap = _seasonsPerYear(); hop <= cap && seasonOfDay(d) !== 'winter'; hop++) d = _nextSeasonDay(d);
  return seasonOfDay(d) === 'winter' ? _nextSeasonDay(d) : p;   // 겨울이 없는 달력이면 종전대로(안전)

```

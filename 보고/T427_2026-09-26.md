## 2. ② 시계 손잡이 — `ZONE_CLOCK_ANCHOR=boot`(계측·하네스 전용 · 기본 끔)

```js
// zone.js 맨 위(어느 require 보다 앞)
const ZONE_CLOCK_ANCHOR = process.env.ZONE_CLOCK_ANCHOR || '';
const _clockReady = (() => {
  if (ZONE_CLOCK_ANCHOR !== 'boot') return null;
  const real = Date.now, t0 = real();
  let at = 0;
  Date.now = function () { return at ? t0 + (real() - at) : t0; };   // 기동 동안 t0 에 멎고, 준비 뒤 t0 에서 실시간 빠르기로
  return () => { if (!at) at = real(); return at - t0; };
})();
// zone.js 맨 끝(파일 평가가 끝난 순간 = 기동 끝 · 어느 타이머 콜백보다 먼저)
if (_clockReady) console.log(`… T427 시계 고정 — … 멎어 있던 벽시계 ${_clockReady()}ms`);
```

* 두 존이 같은 벽시각에 뜨면 기동이 15초든 61초든 **같은 게임 시각(뜬 순간)에 첫 틱**을 돈다(실측: 기동 14.4 · 61.0초를 멎어 있었다 — 판 B).
* 흐르는 빠르기는 그대로 — 간격·타이머·틱은 실시간이다(새 수 0). 끄면 `Date.now` 는 **그 함수 그대로**(⑯-b).

   사냥터 후보를 "그날 서 있는 짐승 자리"(≤24)에서 고르는 것 자체가 뿌리다 — 짐승이 바위 위에 서 있을 수 있다(야생 층 · 회부).

   T146 이 이미 "이 수는 값 판정거리다(재민)"라 적은 자리(이 카드 무접촉). 계측 쪽 규율은 `인계/M-이동입력.md` 자 규율 10(두 팔을 잴 때 다른 하네스를 같이 돌리지 않는다).
